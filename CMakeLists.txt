cmake_minimum_required(VERSION 3.20)
project(mcg_recomp C)

# The lifted code models a 32-bit machine but is portable C, so the host is built
# x64: the target's whole VA range (0x00600000 + ~2 MB of image, then the stack
# and a 256 MB heap arena above it) sits far below anything the host occupies,
# and the 1:1 mapping in recomp_runtime.c holds with g_mem_base at 0.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure for x64; a 32-bit host collides with the mapped image.")
endif()

set(CMAKE_C_STANDARD 11)

# pcrecomp checkout: shared runtime + lifters, a sibling of this repo.
set(PCRECOMP "${CMAKE_SOURCE_DIR}/../tools" CACHE PATH "pcrecomp checkout")
if(NOT EXISTS "${PCRECOMP}/runtime/recomp32/image_loader.c")
  message(FATAL_ERROR "pcrecomp not found at ${PCRECOMP} -- clone "
                      "https://github.com/sp00nznet/pcrecomp next to this repo "
                      "or pass -DPCRECOMP=<path>.")
endif()

# Generated by run_pipeline.py; not in the repo (it is derived from your own
# copy of the game).
file(GLOB GEN_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/recomp/gen/*.c")
if(NOT GEN_SOURCES)
  message(FATAL_ERROR "src/recomp/gen is empty -- run:\n"
                      "  py -3.11 run_pipeline.py\n"
                      "  py -3.11 gen_imports.py")
endif()

add_executable(mcg_recomp
  ${GEN_SOURCES}
  src/runtime/imports_gen.c
  src/runtime/recomp_runtime.c
  src/runtime/shims_impl.c
  src/runtime/ddraw.c
  "${PCRECOMP}/runtime/recomp32/image_loader.c"
  "${PCRECOMP}/runtime/recomp32/crash_report.c"
)

target_include_directories(mcg_recomp PRIVATE
  src/recomp/gen                       # recomp_types.h, recomp_funcs.h
  src/runtime                          # imports.h
  "${PCRECOMP}/runtime/recomp32"       # image_loader.h, crash_report.h
)

# The dummy return address RECOMP_CALL pushes.
#
# 0xDEAD0000 is the canary: easy to spot in a register dump and deliberately
# destructive, so a stack imbalance announces itself instead of quietly becoming
# a zero that some null check swallows. It earned its place immediately -- it
# turned up in esi during CRT startup, which is how the single-instance check
# came to believe another copy of the game was running.
#
# That imbalance is an over-pop inside _initterm's constructor loop: `pop esi`
# retrieves the dummy return address rather than the saved esi, so esp is 4 too
# HIGH, which the call-site balance check cannot see (it looks for esp too low,
# the signature of a dropped `ret`). Finding it is open work.
#
# Default to 0 meanwhile: a zero landing in a register is inert, and
# hPrevInstance == 0 is what Win32 actually passes, so bring-up continues past
# it. This is a MITIGATION, not a fix -- configure with
# -DRECOMP_RETADDR=0xDEAD0000u to put the canary back whenever a value looks
# impossible.
set(RECOMP_RETADDR "0u" CACHE STRING
    "Dummy return address pushed by RECOMP_CALL; 0xDEAD0000u to hunt leaks")
target_compile_definitions(mcg_recomp PRIVATE RECOMP_RETADDR=${RECOMP_RETADDR})

if(MSVC)
  # 829,000 lines of generated C: /MP is the difference between a coffee and a
  # lunch. Nothing here is performance-sensitive enough to want /O2 while the
  # bring-up is still reading stack traces.
  target_compile_options(mcg_recomp PRIVATE /MP /W1)
  # The generated translation units are far past the COFF section limit.
  target_compile_options(mcg_recomp PRIVATE /bigobj)
endif()

target_link_libraries(mcg_recomp PRIVATE kernel32 user32 gdi32 winmm advapi32 dwmapi)
